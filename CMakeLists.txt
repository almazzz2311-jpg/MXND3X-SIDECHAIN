cmake_minimum_required(VERSION 3.22)
project(MXND3XSidechain VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MXND3X_DSP_ONLY "Build portable DSP tests only" OFF)
include(CTest)
add_executable(dsp_tests Tests/DSPTests.cpp)
target_include_directories(dsp_tests PRIVATE Source)
add_test(NAME DSP COMMAND dsp_tests)
if(MXND3X_DSP_ONLY)
  return()
endif()
set(JUCE_DIR "" CACHE PATH "Local JUCE 8.0.6 source checkout")
if(JUCE_DIR)
  add_subdirectory("${JUCE_DIR}" JUCE)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(MXND3XSidechain COMPANY_NAME MXND3X BUNDLE_ID com.mxnd3x.sidechain
  PLUGIN_MANUFACTURER_CODE Mxnd PLUGIN_CODE M3sc FORMATS VST3 Standalone
  PRODUCT_NAME "MXND3X SIDECHAIN" IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE
  NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE VST3_CATEGORIES Fx Dynamics
  COPY_PLUGIN_AFTER_BUILD FALSE)
target_include_directories(MXND3XSidechain PUBLIC Source)
target_sources(MXND3XSidechain PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(MXND3XSidechain PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(MXND3XSidechain PUBLIC juce::juce_audio_utils juce::juce_dsp PRIVATE juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Source/Assets/MXND3XCharacter.png")
  juce_add_binary_data(MXND3XAssets SOURCES Source/Assets/MXND3XCharacter.png)
  target_link_libraries(MXND3XSidechain PRIVATE MXND3XAssets)
  target_compile_definitions(MXND3XSidechain PRIVATE MXND3X_HAS_CHARACTER=1)
endif()
if(BUILD_TESTING)
  juce_add_console_app(processor_tests PRODUCT_NAME "Processor tests")
  target_sources(processor_tests PRIVATE Tests/ProcessorTests.cpp)
  target_link_libraries(processor_tests PRIVATE MXND3XSidechain)
  add_test(NAME Processor COMMAND processor_tests)
endif()
juce_add_gui_app(ui_capture PRODUCT_NAME "MXND3X UI Capture")
target_sources(ui_capture PRIVATE Tests/UICapture.cpp)
target_link_libraries(ui_capture PRIVATE MXND3XSidechain)
